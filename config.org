** Faster Startup

#+begin_src elisp :tangle early-init.el
;; Delay garbage collection while Emacs is booting
(setq gc-cons-threshold most-positive-fixnum
      read-process-output-max (* 4 1024 1024)
      gc-cons-percentage 0.6)

;; Temporarily disable file name handler ALIST during boot
(defvar dj/default-file-name-handler-alist file-name-handler-alist)
(setq file-name-handler-alist nil)

;; Sensible values for after booting
(add-hook 'after-init-hook
          (lambda ()
            (setq gc-cons-threshold (* 100 1024 1024)
                  gc-cons-percentage 0.1
                  file-name-handler-alist dj/default-file-name-handler-alist)))

;; A single VC backend speeds up booting
(setq vc-handled-backends '(Git))

(setq inhibit-compacting-font-caches t)
#+end_src
